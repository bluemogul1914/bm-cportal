    $stmt->execute(['client_id' => $clientId]);
    $ticketCount = $stmt->fetchColumn();
    
    // Count unpaid invoices
    $stmt = $db->prepare("SELECT COUNT(*) FROM invoices WHERE client_id = :client_id AND status = 'unpaid'");
    $stmt->execute(['client_id' => $clientId]);
    $invoiceCount = $stmt->fetchColumn();
    
    // Count active services
    $stmt = $db->prepare("SELECT COUNT(*) FROM subscriptions WHERE client_id = :client_id AND status = 'active'");
    $stmt->execute(['client_id' => $clientId]);
    $serviceCount = $stmt->fetchColumn();
}

// Get recent tickets
$recentTickets = [];
if ($clientId) {
    $stmt = $db->prepare("
        SELECT * FROM tickets 
        WHERE client_id = :client_id 
        AND status IN ('open', 'in_progress')
        ORDER BY created_at DESC 
        LIMIT 5
    ");
    $stmt->execute(['client_id' => $clientId]);
    $recentTickets = $stmt->fetchAll();
}

// Get unpaid invoices
$unpaidInvoices = [];
if ($clientId) {
    $stmt = $db->prepare("
        SELECT * FROM invoices 
        WHERE client_id = :client_id 
        AND status = 'unpaid'
        ORDER BY due_date ASC 
        LIMIT 5
    ");
    $stmt->execute(['client_id' => $clientId]);
    $unpaidInvoices = $stmt->fetchAll();
}

// Get active services
$activeServices = [];
if ($clientId) {
    $stmt = $db->prepare("
        SELECT s.*, p.name as product_name, p.category, p.billing_period 
        FROM subscriptions s
        JOIN products p ON s.product_id = p.id
        WHERE s.client_id = :client_id 
        AND s.status = 'active'
        ORDER BY s.created_at DESC
    ");
    $stmt->execute(['client_id' => $clientId]);
    $activeServices = $stmt->fetchAll();
